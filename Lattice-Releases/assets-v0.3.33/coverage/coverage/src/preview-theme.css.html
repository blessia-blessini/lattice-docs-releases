/* Override github-markdown-css per preview theme.
   Must use the current variable names (--bgColor-*, --fgColor-*) since
   the old names (--color-canvas-*) are no longer used by the library. */
.markdown-body[data-theme="dark"] {
  color-scheme: dark;
  color: #f0f6fc !important;
  --bgColor-default: #0d1117 !important;
  --bgColor-muted: #151b23 !important;
  --fgColor-default: #f0f6fc !important;
  --fgColor-muted: #9198a1 !important;
  --borderColor-default: #3d444d !important;
  --borderColor-muted: #3d444db3 !important;
  --bgColor-neutral-muted: #656c7633 !important;
 
  & table tr    { background-color: #0d1117 !important; }

  & code, & pre { background-color: #151b23 !important; color: #f0f6fc !important; }
